function timeAgo(dateStr) {
  if (!dateStr) return '';
  const d = new Date(dateStr);
  const diff = Math.floor((Date.now() - d) / 1000);
  if (diff < 60) return `${diff}s ago`;
  if (diff < 3600) return `${Math.floor(diff/60)}m ago`;
  if (diff < 86400) return `${Math.floor(diff/3600)}h ago`;
  return d.toLocaleDateString('en-GB', {day:'numeric',month:'short'});
}

async function loadStats() {
  try {
    const r = await fetch(`${API_BASE}/api/stats`);
    if (!r.ok) return;
    const d = await r.json();
    document.getElementById('statCountries').textContent = (d.countries || d.total_countries || '—') + '+';
    document.getElementById('statSources').textContent = (d.sources || d.total_sources || '—');

async function loadTopStories() {
  try {
    const r = await fetch(`${API_BASE}/api/articles/top-stories`);
    if (!r.ok) throw new Error();
    const d = await r.json();
    topStories = Array.isArray(d) ? d : (d.articles || d.data || []);

async function loadArticles(page=1, category=null, search='', country=null) {
  showLoading(true);
  try {
    let url = `${API_BASE}/api/articles?page=${page}&limit=${PAGE_SIZE}`;
    if (category) url += `&category=${encodeURIComponent(category)}`;
    if (search) url += `&search=${encodeURIComponent(search)}`;
    if (country) url += `&country=${encodeURIComponent(country)}`;
    const r = await fetch(url);
    if (!r.ok) throw new Error();
    const d = await r.json();
    allArticles = Array.isArray(d) ? d : (d.articles || d.data || []);
    const total = d.total || d.total_count || allArticles.length;
    const totalPages = d.totalPages || d.total_pages || Math.ceil(total / PAGE_SIZE);

    if (search) {
      renderGrid(allArticles, 'searchGrid');
      document.getElementById('searchCountDisplay').textContent = `${total.toLocaleString()} result${total !== 1 ? 's' : ''}`;
      renderPagination('searchPagination', page, totalPages, (p) => {
        currentPage = p;
        loadArticles(p, null, search, null);
      });
    } else if (category) {
      renderGrid(allArticles, 'catArticlesGrid');
      document.getElementById('catArticlesCount').textContent = `${total.toLocaleString()} articles`;
      renderPagination('catPagination', page, totalPages, (p) => {
        currentPage = p;
        loadArticles(p, category, '', null);
        window.scrollTo({top:0,behavior:'smooth'});
      });
    } else {
      renderGrid(allArticles, 'articlesGrid');
      document.getElementById('articlesCount').textContent = `${total.toLocaleString()} articles`;
      renderPagination('pagination', page, totalPages, (p) => {
        currentPage = p;
        loadArticles(p, null, '', null);
        window.scrollTo({top:0,behavior:'smooth'});
      });
    }
  } catch(e) {
    const target = search ? 'searchGrid' : (category ? 'catArticlesGrid' : 'articlesGrid');
    document.getElementById(target).innerHTML = `<div class="empty-state" style="grid-column:1/-1"><div class="empty-icon">📡</div><h3>Connection issue</h3><p>Could not load articles. Please refresh.</p></div>`;
  } finally {
    showLoading(false);
  }
}

// ─── PAGINATION ───
function renderPagination(containerId, current, total, onPage) {
  const el = document.getElementById(containerId);
  if (!el || total <= 1) { if(el) el.innerHTML = ''; return; }
  
  const pages = [];
  if (total <= 7) {
    for (let i = 1; i <= total; i++) pages.push(i);
  } else {
    pages.push(1);
    if (current > 3) pages.push('...');
    for (let i = Math.max(2, current-1); i <= Math.min(total-1, current+1); i++) pages.push(i);
    if (current < total - 2) pages.push('...');
    pages.push(total);
  }

  el.innerHTML = `
    <button class="page-btn" onclick="(${onPage.toString()})(${current - 1})" ${current === 1 ? 'disabled' : ''}>
      <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round"><path d="m15 18-6-6 6-6"/></svg>
    </button>
    ${pages.map(p => p === '...' 
      ? `<span class="page-btn" style="cursor:default;border:none">…</span>`
      : `<button class="page-btn ${p === current ? 'active' : ''}" onclick="(${onPage.toString()})(${p})">${p}</button>`
    ).join('')}
    <button class="page-btn" onclick="(${onPage.toString()})(${current + 1})" ${current === total ? 'disabled' : ''}>
      <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round"><path d="m9 18 6-6-6-6"/></svg>
    </button>
  `;
}

// ─── REGION COUNTS (best effort from loaded articles) ───
function populateRegionCounts() {
  // Will update once articles are loaded
  setTimeout(() => {
    const counts = {'North Africa':0,'West Africa':0,'East Africa':0,'Central Africa':0,'Southern Africa':0};
    if (allArticles.length === 0 && topStories.length === 0) return;
    const combined = [...topStories, ...allArticles];
    combined.forEach(a => {
      const region = getRegion(a.country);
      if (counts[region] !== undefined) counts[region]++;
    });
    const idMap = {'North Africa':'rc-north','West Africa':'rc-west','East Africa':'rc-east','Central Africa':'rc-central','Southern Africa':'rc-south'};
    Object.entries(counts).forEach(([region, count]) => {
      const el = document.getElementById(idMap[region]);
      if (el && count > 0) el.textContent = count + ' stories';
    });
  }, 2000);
}

// ─── NEWSLETTER ───
async function subscribeNewsletter() {
  const email = document.getElementById('newsletterEmail').value.trim();
  if (!email || !email.includes('@')) return;
  try {
    await fetch(`${API_BASE}/api/newsletter/subscribe`, {
      method: 'POST',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({email})
    });
  } catch(e) {}
  document.querySelector('.newsletter-form').style.display = 'none';
  document.getElementById('newsletterSuccess').style.display = 'block';
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str)
    .replace(/&/g,'&amp;')
    .replace(/</g,'&lt;')
    .replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;')
    .replace(/'/g,'&#39;');
}
